# Separate from pyproject.toml/.venv on purpose: databricks-connect and
# pyspark cannot coexist in the same environment (databricks-connect ships
# its own pyspark-compatible client). Use venv_dbc for anything that needs
# a live Spark session against a Databricks workspace (interactive
# exploration, notebook-style scripts); use .venv for everything else
# (ingestion unit tests, dbt, local pyspark logic).
#
# Targets serverless compute. Pinned (not just a floor) because newer
# databricks-connect releases can outpace what this workspace's serverless
# backend supports - e.g. 19.0.0 fails with "Serverless mode is not yet
# supported in this version of Databricks Connect". 16.1.7 is confirmed
# working here; bump deliberately and re-run scripts/dbc_connection_check.py
# before trusting a newer version.
databricks-connect==16.1.7
databricks-sdk>=0.20
python-dotenv>=1.0
